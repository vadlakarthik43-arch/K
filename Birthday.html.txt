  <script>
    // 1. Generate Background Stars and Balloons
    const bgContainer = document.getElementById('bgDecorations');
    
    // Create Stars
    for (let i = 0; i < 40; i++) {
      const star = document.createElement('div');
      star.className = 'star';
      star.style.left = `${Math.random() * 100}%`;
      star.style.top = `${Math.random() * 100}%`;
      const size = Math.random() * 3 + 1;
      star.style.width = `${size}px`;
      star.style.height = `${size}px`;
      star.style.setProperty('--duration', `${Math.random() * 3 + 2}s`);
      bgContainer.appendChild(star);
    }

    // Create Floating Balloons
    const colors = ['#ff4081', '#7c4dff', '#00e676', '#ffea00', '#00b0ff'];
    for (let i = 0; i < 12; i++) {
      const balloon = document.createElement('div');
      balloon.className = 'balloon';
      balloon.style.left = `${Math.random() * 90 + 5}%`;
      balloon.style.setProperty('--color', colors[Math.floor(Math.random() * colors.length)]);
      balloon.style.setProperty('--speed', `${Math.random() * 8 + 7}s`);
      balloon.style.animationDelay = `${Math.random() * 5}s`;
      bgContainer.appendChild(balloon);
    }

    // 2. Fireworks / Confetti Particle System
    const canvas = document.getElementById('effectCanvas');
    const ctx = canvas.getContext('2d');
    let particles = [];

    function resizeCanvas() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    class Particle {
      constructor(x, y) {
        this.x = x;
        this.y = y;
        this.size = Math.random() * 7 + 3;
        this.speedX = (Math.random() - 0.5) * 10;
        this.speedY = (Math.random() - 0.5) * 10 - 2;
        this.color = colors[Math.floor(Math.random() * colors.length)];
        this.gravity = 0.12;
        this.opacity = 1;
        this.decay = Math.random() * 0.015 + 0.005;
      }

      update() {
        this.speedY += this.gravity;
        this.x += this.speedX;
        this.y += this.speedY;
        this.opacity -= this.decay;
      }

      draw() {
        ctx.save();
        ctx.globalAlpha = this.opacity;
        ctx.fillStyle = this.color;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      }
    }

    function triggerFireworks() {
      const centerX = window.innerWidth / 2;
      const centerY = window.innerHeight / 2;
      for (let i = 0; i < 120; i++) {
        particles.push(new Particle(centerX, centerY));
      }
      setTimeout(() => {
        for (let i = 0; i < 80; i++) {
          particles.push(new Particle(window.innerWidth * 0.2, window.innerHeight * 0.3));
        }
      }, 200);
      setTimeout(() => {
        for (let i = 0; i < 80; i++) {
          particles.push(new Particle(window.innerWidth * 0.8, window.innerHeight * 0.3));
        }
      }, 400);
    }

    function animateParticles() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].update();
        particles[i].draw();
        if (particles[i].opacity <= 0) {
          particles.splice(i, 1);
        }
      }
      requestAnimationFrame(animateParticles);
    }
    animateParticles();

    // 3. Step Transitions
    function switchStep(fromId, toId) {
      document.getElementById(fromId).classList.remove('active');
      document.getElementById(toId).classList.add('active');
    }

    function openGift() {
      switchStep('step1', 'step2');
    }

    function submitName() {
      const nameInput = document.getElementById('userNameInput').value.trim();
      if (nameInput) {
        document.getElementById('greetingName').innerText = `Happy Birthday, ${nameInput}! 🎉`;
      }
      switchStep('step2', 'step3');
    }

    function startCelebration() {
      triggerFireworks();
      switchStep('step3', 'step4');
    }

    function goToStep5() {
      switchStep('step4', 'step5');
    }

    function showLoveWish() {
      triggerFireworks();
      switchStep('step5', 'step6');
    }

    function goToEnd() {
      switchStep('step6', 'step7');
    }
  </script>
